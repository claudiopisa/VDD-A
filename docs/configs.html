 11class DefaultConfigLoader:
 12    """
 13    Loads default configuration instances based on the config class name.
 14    
 15    Maps config classes to their corresponding default config classes:
 16    - CoreConfig -> CoreDefaultConfig
 17    - FileVersioningConfig -> FileVersioningDefaultConfig
 18    - TaskVersioningConfig -> TaskVersioningDefaultConfig
 19    """
 20    
 21    # Registry to store manual mappings if needed
 22    _registry: Dict[str, Type] = {}
 23    
 24    @classmethod
 25    def register(cls, config_class_name: str, default_config_class: Type):
 26        """
 27        Manually register a mapping between a config class and its default config.
 28        
 29        Args:
 30            config_class_name: Name of the config class (e.g., 'CoreConfig')
 31            default_config_class: The default config class to instantiate
 32        """
 33        cls._registry[config_class_name] = default_config_class
 34    
 35    @classmethod
 36    def load(cls, config_instance: Any) -> Any:
 37        """
 38        Load and return the appropriate default config instance.
 39        
 40        Args:
 41            config_instance: Instance of a Config subclass
 42            
 43        Returns:
 44            Instance of the corresponding default config class
 45            
 46        Raises:
 47            ImportError: If the default config class cannot be imported
 48            ValueError: If no default config exists for the given class
 49        """
 50        config_class_name = config_instance.__class__.__name__
 51        
 52        # Check if manually registered
 53        if config_class_name in cls._registry:
 54            default_config_class = cls._registry[config_class_name] # returns the actual class object, of type `type`
 55            return default_config_class()
 56        
 57        # Auto-resolve based on naming convention
 58        return cls._auto_resolve(config_class_name)
 59    
 60    @classmethod
 61    def _auto_resolve(cls, config_class_name: str) -> Any:
 62        """
 63        Automatically resolve the default config class based on naming convention.
 64        
 65        Convention: XyzConfig -> XyzDefaultConfig
 66        
 67        Args:
 68            config_class_name: Name of the config class
 69            
 70        Returns:
 71            Instance of the default config class
 72            
 73        Raises:
 74            ImportError: If the default config module cannot be imported
 75            ValueError: If the config class doesn't follow naming conventions
 76        """
 77        # Remove 'Config' suffix and convert to snake_case for module name
 78        if not config_class_name.endswith('Config'):
 79            raise ValueError(
 80                f"Config class '{config_class_name}' must end with 'Config'"
 81            )
 82        
 83        # Extract the base name (e.g., 'Core' from 'CoreConfig')
 84        base_name = config_class_name[:-6]  # Remove 'Config'
 85        
 86        # Convert to snake_case for module name
 87        module_name = cls._camel_to_snake(base_name) + '_default_config'
 88        
 89        # Construct default config class name
 90        default_class_name = base_name + 'DefaultConfig'
 91        
 92        # Dynamic import
 93        try:
 94            module = __import__(
 95                f'configs.default_config.{module_name}',
 96                fromlist=[default_class_name]
 97            )
 98            default_config_class = getattr(module, default_class_name)
 99            return default_config_class()
100        except (ImportError, AttributeError) as e:
101            raise ImportError(
102                f"Could not load default config for '{config_class_name}'. "
103                f"Expected class '{default_class_name}' in module "
104                f"'configs.default_config.{module_name}'. Error: {e}"
105            )
106    
107    @staticmethod
108    def _camel_to_snake(name: str) -> str:
109        """
110        Convert CamelCase to snake_case.
111        
112        Args:
113            name: CamelCase string
114            
115        Returns:
116            snake_case string
117        """
118        # Insert underscore before uppercase letters and convert to lowercase
119        s1 = re.sub('(.)([A-Z][a-z]+)', r'\1_\2', name)
120        return re.sub('([a-z0-9])([A-Z])', r'\1_\2', s1).lower()
